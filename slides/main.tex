\begin{frame}{Семь методов оценивают одну полезность}
\small
\centering
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{lp{9cm}}
\toprule
Метод & Обучающая цель / представление \\
\midrule
Direct & Прямая регрессия прироста \\
Marginal & Две оценки правильности \\
Events & Спасение и вред отдельными головами \\
Ternary & Три значения прироста \\
Joint & Четыре совместных исхода экспертов \\
Hurdle & Наличие расхождения и его направление \\
Soft-gain & Мягкая метка прироста \\
Weak-error & Ошибка первого эксперта: отдельная proxy-цель \\
\bottomrule
\end{tabular}
\medskip
\takeaway{Различаем выбор семейства цели и выбор параметризации внутри gain.}
\source{Confidence и random — необучаемые контроли. Основания: Jitkrittum et al., 2023; Nyberg \& Klami, 2023.}
\end{frame}

\begin{frame}{Нужно оценивать спасения за вычетом вреда}
\begin{columns}[T]
\begin{column}{.49\textwidth}
\centering\small
\textbf{Парная разметка исходов}\par\medskip
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.7}
\begin{tabular}{lcc}
\toprule
 & Второй: 0 & Второй: 1 \\
\midrule
Первый: 0 & 0 & \textcolor{accent}{+1: спасение} \\
Первый: 1 & \textcolor{warm}{−1: вред} & 0 \\
\bottomrule
\end{tabular}
\source{0 — ошибка; 1 — правильный ответ.}
\end{column}
\begin{column}{.46\textwidth}
\small Первый исход $W$, второй $S$; $D=S-W$.
\[
g(z)=\mathbb E[D\mid z]
\]
\centering
\begin{tikzpicture}[node distance=.65cm]
\node[box] (a) {Оценить $g(z)$};
\node[box,below=of a] (b) {Выбрать top-$k$};
\draw[arr] (a)--(b);
\end{tikzpicture}
\end{column}
\end{columns}
\bigskip
\takeaway{При ровно $k$ вызовах top-$k$ максимизирует ожидаемое качество.}
\source{Признаки доступны до второго вызова. Равная квота, не измеренный денежный бюджет.}
\end{frame}
